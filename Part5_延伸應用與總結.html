⏰ 延伸 1：設定自動排程推播

互動模式下，使用者隨時發訊息就能查詢。但如果你想每天自動推送固定地點的日報，可以加上排程功能：

操作步驟

  1. 在 GAS 編輯器中，點擊左側的 ⏰ 圖示（觸發條件）
  2. 點擊「+ 新增觸發條件」
  3. 設定以下選項：
    • 選擇要執行的函式：dailyPush
    • 選擇事件來源：「以時間驅動」
    • 選擇時間型觸發條件類型：「日計時器」
    • 選擇時段：「上午 8 點到 9 點」
  4. 點擊「儲存」
設定完成！每天早上 8 點，你的 LINE 就會自動收到衛星觀測日報！完全不用動手！

進階：用程式設定觸發條件

// === ⏰ 每日自動推送（額外功能）===
function dailyPush() {
  const USER_ID = '貼上你的 User ID';
  const data = getSatelliteData(23.0, 120.2);
  const message = interpretData(data, '台南');

  const url = 'https://api.line.me/v2/bot/message/push';
  const payload = {
    to: USER_ID,
    messages: [{ type: 'text', text: message }]
  };
  UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + LINE_TOKEN },
    payload: JSON.stringify(payload)
  });
}

function setDailyTrigger() {
  ScriptApp.getProjectTriggers().forEach(t =>
    ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('dailyPush')
    .timeBased().everyDays(1).atHour(8).create();
  Logger.log('✅ 已設定每日自動推播！');
}
💡 排程推播使用的是 Push API（主動推送），跟互動回覆用的 Reply API 不同。Push API 需要 User ID。

🌱 延伸 3：進階 — NDVI 植被監測

如果你想更進一步，可以使用 Sentinel Hub 取得 NDVI 植被指數資料

核心概念

  • NDVI（植被指數）是衛星影像分析的核心指標
  • Sentinel Hub 提供免費試用帳號（30,000 處理單位）
  • 可以判斷特定區域「有沒有變綠 / 變乾」

🌾 農業監測

追蹤作物生長狀況

🌊 水體變化

監測洪水、乾旱

🏙️ 城市熱島

搭配溫度資料分析

🔥 森林異常

火災前後比較

Sentinel Hub 註冊網址：https://www.sentinel-hub.com/ （免費試用）
⚠️ NDVI 需要較多的設定步驟，建議作為「進階自學」項目

// ===================================
// 🛰️ 低軌衛星資訊 × LINE 互動機器人
// 作者：阿亮老師教學專案
// ===================================

// === 📌 設定區 ===
const LINE_TOKEN = '貼上你的 Channel Access Token';

const LOCATIONS = {
  '台北': { lat: 25.0, lon: 121.5 },
  '台中': { lat: 24.1, lon: 120.7 },
  '台南': { lat: 23.0, lon: 120.2 },
  '高雄': { lat: 22.6, lon: 120.3 },
  '花蓮': { lat: 24.0, lon: 121.6 },
  '墾丁': { lat: 21.9, lon: 120.8 },
  '東京': { lat: 35.7, lon: 139.7 },
  '紐約': { lat: 40.7, lon: -74.0 }
};

// === 📨 Webhook 入口 ===
function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const events = data.events;
  for (let i = 0; i < events.length; i++) {
    if (events[i].type === 'message' && events[i].message.type === 'text') {
      handleTextMessage(events[i]);
    }
  }
  return ContentService.createTextOutput('OK');
}

// === 🤖 處理使用者訊息 ===
function handleTextMessage(event) {
  const text = event.message.text.trim();
  const replyToken = event.replyToken;

  if (LOCATIONS[text]) {
    const loc = LOCATIONS[text];
    const data = getSatelliteData(loc.lat, loc.lon);
    const message = interpretData(data, text);
    logToSheet(text, loc.lat, loc.lon, data);
    replyToLINE(replyToken, message);
    return;
  }

  const coordMatch = text.match(/^([\d.-]+)\s*[,，\s]\s*([\d.-]+)$/);
  if (coordMatch) {
    const lat = parseFloat(coordMatch[1]);
    const lon = parseFloat(coordMatch[2]);
    const data = getSatelliteData(lat, lon);
    const message = interpretData(data, lat + '°N, ' + lon + '°E');
    logToSheet(lat + '°N, ' + lon + '°E', lat, lon, data);
    replyToLINE(replyToken, message);
    return;
  }

  replyToLINE(replyToken, getHelpMessage());
}

// === ❓ 使用說明 ===
function getHelpMessage() {
  return '🛰️ 低軌衛星觀測機器人\n' +
    '━━━━━━━━━━━━\n' +
    '📍 輸入城市名稱，例如：\n' +
    '   台北、台中、台南、高雄\n\n' +
    '📐 或輸入座標，例如：\n' +
    '   23.0, 120.2\n\n' +
    '🌏 支援全球任何地點！';
}

// === 📤 回覆 LINE ===
function replyToLINE(replyToken, message) {
  UrlFetchApp.fetch('https://api.line.me/v2/bot/message/reply', {
    method: 'post',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + LINE_TOKEN },
    payload: JSON.stringify({
      replyToken: replyToken,
      messages: [{ type: 'text', text: message }]
    })
  });
}

// === 🛰️ 取得衛星資料 ===
function getSatelliteData(lat, lon) {
  const today = new Date();
  const end = Utilities.formatDate(today, 'GMT+8', 'yyyyMMdd');
  const weekAgo = new Date(today.getTime() - 7 * 24 * 60 * 60 * 1000);
  const start = Utilities.formatDate(weekAgo, 'GMT+8', 'yyyyMMdd');

  const url = 'https://power.larc.nasa.gov/api/temporal/daily/point' +
    '?parameters=T2M,T2M_MAX,T2M_MIN,PRECTOTCORR,RH2M,ALLSKY_SFC_SW_DWN,WS2M' +
    '&community=AG&longitude=' + lon + '&latitude=' + lat +
    '&start=' + start + '&end=' + end + '&format=JSON';

  return JSON.parse(UrlFetchApp.fetch(url).getContentText());
}

// === 📊 解讀資料（含分析建議）===
function interpretData(data, locationName) {
  const params = data.properties.parameter;
  const dates = Object.keys(params.T2M);
  const latestDate = dates[dates.length - 1];

  const temp = params.T2M[latestDate];
  const tempMax = params.T2M_MAX[latestDate];
  const tempMin = params.T2M_MIN[latestDate];
  const rain = params.PRECTOTCORR[latestDate];
  const humidity = params.RH2M[latestDate];
  const solar = params.ALLSKY_SFC_SW_DWN[latestDate];
  const wind = params.WS2M[latestDate];

  let tempStatus = temp > 30 ? '高溫炎熱 🔥' :
    temp > 20 ? '溫暖舒適 ☀️' :
    temp > 10 ? '涼爽 🍂' : '寒冷 ❄️';

  let rainStatus = rain > 10 ? '有明顯降雨 🌧️' :
    rain > 1 ? '少量降雨 🌦️' : '無降雨 ☀️';

  let windStatus = wind > 10 ? '強風 💨' :
    wind > 5 ? '微風 🍃' : '無風 🌤️';

  let analysis = [];
  if (temp <= 15) analysis.push('氣溫偏低，建議穿著保暖衣物');
  else if (temp <= 22) analysis.push('氣溫涼爽，建議攜帶薄外套');
  else if (temp >= 33) analysis.push('高溫警戒，注意防曬補水');
  if (tempMax - tempMin > 10) analysis.push('日夜溫差大，早晚記得加衣');
  if (rain > 10) analysis.push('降雨明顯，請攜帶雨具');
  else if (rain > 1) analysis.push('可能有零星降雨，建議帶傘');
  else analysis.push('無降雨，適合戶外活動');
  if (humidity > 80) analysis.push('濕度偏高，體感較悶熱');
  if (solar > 20) analysis.push('日照強烈，注意防曬');
  if (wind > 10) analysis.push('風速較大，戶外活動注意安全');

  return '\n🛰️ 衛星觀測日報' +
    '\n━━━━━━━━━━━━' +
    '\n📍 地點：' + locationName +
    '\n📅 日期：' + latestDate +
    '\n━━━━━━━━━━━━' +
    '\n🌡️ 氣溫：' + temp.toFixed(1) + '°C（' + tempStatus + '）' +
    '\n📊 最高/最低：' + tempMax.toFixed(1) + '°C / ' + tempMin.toFixed(1) + '°C' +
    '\n💧 降雨量：' + rain.toFixed(1) + ' mm（' + rainStatus + '）' +
    '\n💨 相對濕度：' + humidity.toFixed(0) + '%' +
    '\n☀️ 日照量：' + solar.toFixed(1) + ' MJ/m²' +
    '\n🌬️ 風速：' + wind.toFixed(1) + ' m/s（' + windStatus + '）' +
    '\n━━━━━━━━━━━━' +
    '\n💬 衛星資料分析：' +
    '\n' + analysis.map(a => '• ' + a).join('\n') +
    '\n━━━━━━━━━━━━' +
    '\n📡 資料來源：NASA POWER（低軌衛星觀測）';
}

// === 📝 記錄查詢結果到 Google Sheet ===
function logToSheet(locationName, lat, lon, data) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const params = data.properties.parameter;
  const dates = Object.keys(params.T2M);
  const latestDate = dates[dates.length - 1];

  sheet.appendRow([
    new Date(),
    locationName + ' (' + lat + ', ' + lon + ')',
    params.T2M[latestDate],
    params.PRECTOTCORR[latestDate],
    params.RH2M[latestDate],
    params.ALLSKY_SFC_SW_DWN[latestDate],
    params.WS2M[latestDate],
    'LINE 使用者'
  ]);
}

// === 🧪 測試 ===
function testAPI() {
  const data = getSatelliteData(23.0, 120.2);
  Logger.log(interpretData(data, '台南'));
}
